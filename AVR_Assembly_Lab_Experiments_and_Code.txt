AVR ASSEMBLY LAB: EXPERIMENT NAMES AND CODE
Target MCU: ATmega328P
Toolchain: MPLAB X IDE with AVR-GCC

Save each experiment in a separate .S file (capital S). The capital S enables
the C preprocessor, which expands <avr/io.h> and the _SFR_* register macros.
Build one experiment at a time in a normal AVR-GCC/MPLAB X project with the
standard startup code enabled. Each source file defines main.

===============================================================================
EXPERIMENT 1: LED ON/OFF USING GPIO
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
main:
    ; Configure PB0 as output
    sbi _SFR_IO_ADDR(DDRB), DDB0

    ; Make PB0 HIGH
    sbi _SFR_IO_ADDR(PORTB), PORTB0

LOOP:
    rjmp LOOP


===============================================================================
EXPERIMENT 2: LED BLINKING USING SOFTWARE DELAY
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
main:
    sbi _SFR_IO_ADDR(DDRB), DDB0

MAIN:
    sbi _SFR_IO_ADDR(PORTB), PORTB0
    rcall DELAY
    cbi _SFR_IO_ADDR(PORTB), PORTB0
    rcall DELAY
    rjmp MAIN

DELAY:
    ldi r18, 40
D1:
    ldi r19, 255
D2:
    ldi r20, 255
D3:
    dec r20
    brne D3
    dec r19
    brne D2
    dec r18
    brne D1
    ret


===============================================================================
EXPERIMENT 3: PUSH BUTTON CONTROLS LED
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
main:
    sbi _SFR_IO_ADDR(DDRB), DDB0
    cbi _SFR_IO_ADDR(DDRD), DDD2
    sbi _SFR_IO_ADDR(PORTD), PORTD2

MAIN:
    ; Skip next instruction if PD2 is LOW
    sbic _SFR_IO_ADDR(PIND), PIND2
    rjmp BUTTON_NOT_PRESSED

BUTTON_PRESSED:
    sbi _SFR_IO_ADDR(PORTB), PORTB0
    rjmp MAIN

BUTTON_NOT_PRESSED:
    cbi _SFR_IO_ADDR(PORTB), PORTB0
    rjmp MAIN

EXPERIMENT 4: RUNNING LED PATTERN
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
main:
    ldi r16, 0x0F
    out _SFR_IO_ADDR(DDRB), r16
    ldi r16, 0x01

MAIN:
    out _SFR_IO_ADDR(PORTB), r16
    rcall DELAY
    lsl r16
    cpi r16, 0x10
    brne MAIN
    ldi r16, 0x01
    rjmp MAIN

DELAY:
    ldi r18, 30
D1:
    ldi r19, 255
D2:
    ldi r20, 255
D3:
    dec r20
    brne D3
    dec r19
    brne D2
    dec r18
    brne D1
    ret


===============================================================================
EXPERIMENT 5: PWM GENERATION USING TIMER0
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
main:
    ; PD6 / OC0A output
    sbi _SFR_IO_ADDR(DDRD), DDD6

    ; Fast PWM, non-inverting
    ldi r16, (1<<COM0A1) | (1<<WGM01) | (1<<WGM00)
    out _SFR_IO_ADDR(TCCR0A), r16

    ; Clock / 64
    ldi r16, (1<<CS01) | (1<<CS00)
    out _SFR_IO_ADDR(TCCR0B), r16

    ; Approximately 50% duty cycle
    ldi r16, 128
    out _SFR_IO_ADDR(OCR0A), r16

MAIN:
    rjmp MAIN


===============================================================================
EXPERIMENT 6: ADC CONTROLS LED BRIGHTNESS
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
main:
    ; PWM setup
    sbi _SFR_IO_ADDR(DDRD), DDD6
    ldi r16, (1<<COM0A1) | (1<<WGM01) | (1<<WGM00)
    out _SFR_IO_ADDR(TCCR0A), r16
    ldi r16, (1<<CS01) | (1<<CS00)
    out _SFR_IO_ADDR(TCCR0B), r16

    ; ADC setup: AVcc, left adjusted, ADC0
    ldi r16, (1<<REFS0) | (1<<ADLAR)
    sts _SFR_MEM_ADDR(ADMUX), r16
    ldi r16, (1<<ADEN) | (1<<ADPS2) | (1<<ADPS1) | (1<<ADPS0)
    sts _SFR_MEM_ADDR(ADCSRA), r16

MAIN:
    ; Start ADC conversion
    lds r16, _SFR_MEM_ADDR(ADCSRA)
    ori r16, (1<<ADSC)
    sts _SFR_MEM_ADDR(ADCSRA), r16

WAIT_ADC:
    lds r16, _SFR_MEM_ADDR(ADCSRA)
    ; Wait while ADSC = 1
    sbrc r16, ADSC
    rjmp WAIT_ADC

    ; Read 8-bit ADC result
    lds r17, _SFR_MEM_ADDR(ADCH)

    ; ADC directly controls PWM
    out _SFR_IO_ADDR(OCR0A), r17
    rjmp MAIN


===============================================================================
EXPERIMENT 7: INTERRUPT + ADC + PWM INTEGRATED PROJECT
===============================================================================

#include <avr/io.h>

.text
.global main
.type main, @function
.global __vector_1
.type __vector_1, @function

/* AVR-GCC startup code initializes the stack and routes INT0 to __vector_1. */
main:

    ; PWM setup on PD6 / OC0A
    sbi _SFR_IO_ADDR(DDRD), DDD6
    ldi r16, (1<<COM0A1) | (1<<WGM01) | (1<<WGM00)
    out _SFR_IO_ADDR(TCCR0A), r16
    ldi r16, (1<<CS01) | (1<<CS00)
    out _SFR_IO_ADDR(TCCR0B), r16

    ; ADC setup: AVcc, left adjusted, ADC0
    ldi r16, (1<<REFS0) | (1<<ADLAR)
    sts _SFR_MEM_ADDR(ADMUX), r16
    ldi r16, (1<<ADEN) | (1<<ADPS2) | (1<<ADPS1) | (1<<ADPS0)
    sts _SFR_MEM_ADDR(ADCSRA), r16

    ; Button on PD2 / INT0
    cbi _SFR_IO_ADDR(DDRD), DDD2
    sbi _SFR_IO_ADDR(PORTD), PORTD2
    ldi r16, (1<<ISC01)
    sts _SFR_MEM_ADDR(EICRA), r16
    sbi _SFR_IO_ADDR(EIMSK), INT0

    ; Initial state = ON
    ldi r20, 1
    sei

MAIN:
    tst r20
    breq SYSTEM_OFF

    ; Start ADC conversion
    lds r16, _SFR_MEM_ADDR(ADCSRA)
    ori r16, (1<<ADSC)
    sts _SFR_MEM_ADDR(ADCSRA), r16

WAIT_ADC:
    lds r16, _SFR_MEM_ADDR(ADCSRA)
    sbrc r16, ADSC
    rjmp WAIT_ADC

    ; ADC to PWM
    lds r17, _SFR_MEM_ADDR(ADCH)
    out _SFR_IO_ADDR(OCR0A), r17
    rjmp MAIN

SYSTEM_OFF:
    clr r16
    out _SFR_IO_ADDR(OCR0A), r16
    rjmp MAIN

__vector_1:
    ; Save R16
    push r16

    ; Save status register
    in r16, _SFR_IO_ADDR(SREG)
    push r16

    ; Toggle system state
    ldi r16, 1
    eor r20, r16

    ; Restore SREG
    pop r16
    out _SFR_IO_ADDR(SREG), r16

    ; Restore R16
    pop r16
    reti
